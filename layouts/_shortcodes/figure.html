{{- /* A responsive figure served from the media bucket. Neither this repository nor the application's carries an image binary: screenshots are generated by the workbench pipeline, uploaded to R2 under a content-hashed key, and described here by a small manifest in data/media/. That keeps both repositories small however often the interface changes, and lets a replaced screenshot keep an immutable cache lifetime because its key moves with its content. {{< figure name="vectorscope-cast" >}} {{< figure name="beside-lightroom" frame="plain" caption="..." >}} `alt` and the default caption come from the manifest, where they are written beside the shot the pipeline produced; a `caption` given here overrides it, for when the point being made is the page's rather than the picture's. */ -}} {{- $name := .Get "name" -}} {{- $entry := index hugo.Data.media $name -}} {{- if not $entry -}} {{- /* No manifest yet: the pages read correctly without their figures, so a missing shot is a warning while the pipeline is being brought up and a hard failure once the site is being published for real. */ -}} {{- $msg := printf "figure %q: no manifest at data/media/%s.json — run the workbench screenshot pipeline" $name $name -}} {{- if and hugo.IsProduction (not site.Params.allowMissingFigures) }}{{ errorf "%s" $msg }}{{ else }}{{ warnf "%s" $msg }}{{ end -}} {{- else -}} {{- $base := printf "%s/%s" (partial "media-base.html" .) $entry.base -}} {{- /* Data-file numbers arrive as float64, so they are cast before they reach a width or a %d. */ -}} {{- $widths := apply $entry.variants "int" "." -}} {{- $largest := index $widths (sub (len $widths) 1) -}} {{- $sizes := .Get "sizes" | default "(min-width: 56rem) 40rem, 100vw" -}} {{- $frame := .Get "frame" | default "scope" -}} {{- $caption := .Get "caption" | default $entry.caption -}} {{- end -}}
