{{- /* Markdown images, upgraded. The documentation under content/docs/ is generated from the application repository and has to render in TWO places: here, and in GitHub's markdown preview when somebody reads docs/SCOPES.md on the repository page. A Hugo shortcode would show up as literal text there, so those files use ordinary markdown pointing at the media bucket's STABLE address: ![alt text](https://media.sidescopes.org/shots/histogram.png) GitHub renders that directly. This hook recognises it, finds the shot in data/media/, and replaces it with a full carrying AVIF and WebP at several widths — so the site gets responsive images from the same one line of markdown, with no second source to maintain. Anything that is not a known shot falls through to a plain . */ -}} {{- $destination := .Destination -}} {{- $name := "" -}} {{- if strings.Contains $destination "/shots/" -}} {{- $file := index (last 1 (split $destination "/shots/")) 0 -}} {{- $name = strings.TrimSuffix ".png" $file -}} {{- end -}} {{- $entry := "" -}} {{- if $name }}{{ $entry = index hugo.Data.media $name }}{{ end -}} {{- if $entry -}} {{- $base := printf "%s/%s" (partial "media-base.html" .) $entry.base -}} {{- $widths := apply $entry.variants "int" "." -}} {{- $largest := index $widths (sub (len $widths) 1) -}} {{- $alt := or .Text $entry.alt -}} {{- else -}} {{- end -}}
